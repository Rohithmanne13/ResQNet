# Load the helpers once per terminal. They print "HTTP <code>" + the JSON answer of every call:
#   call METHOD PATH [JSON]   client API      ccall ...  coordinator (internal API)
#   last_id  field  cluster  queue  prom      (see scripts/cli.sh)

$ cd /mnt/c/Users/rohit/Downloads/Cloud_Computing_Project/Emergency-Response-Matching
$ source scripts/cli.sh compose

# ---- 2.1 System configuration: 1 coordinator, 2-3 microservices, 1 RabbitMQ, 3 shards x 2 (PDF §2, §4)
$ call GET /health                         # request-service up and connected to RabbitMQ
$ ccall GET /health                        # coordinator: healthy_shards [0, 1, 2], 6 nodes
$ cluster                                  # 3 shards, each primary + replica, all "ready", equal record counts
$ queue                                    # RabbitMQ queue matching.requests: 2 consumers (2 matching workers)

# ---- 2.2 registerResource(type, location, availability) - POST /resources -> 201      (PDF §1, §3.2, §5)
$ call POST /resources '{"type":"water","kind":"ngo","name":"Kochi NGO water tanker","availability":2,"location":{"lat":9.9400,"lon":76.2700,"area":"Ernakulam"}}'
$ RES=$(last_id); echo $RES                # res-water-...: unique id, type, location, availability, status
$ call POST /resources '{"type":"transportation","kind":"volunteer","name":"Volunteer boat","availability":1,"location":{"lat":9.9350,"lon":76.2650}}'
$ call GET /resources/$RES                 # read it back
$ call GET "/resources?type=transportation&status=available" | head -25    # GET /resources -> 200 (list, filters)

# ---- 2.3 createRequest(type, location, details) + getRequest(requestId)         (PDF §1, §3.1, §5)
$ call POST /requests '{"type":"water","details":"Flooded house: family of 4 needs drinking water","priority":"high","auto_match":false,"location":{"lat":9.9312,"lon":76.2673,"area":"Kochi"}}'
$ REQ=$(last_id); echo $REQ                # req-...: id, type, location, details, status pending, created_at
$ call GET /requests/$REQ                  # GET /requests/{id} -> 200; _storage shows shard + serving node

# ---- 2.4 matchRequest(requestId) - POST /match/{id} -> 200   (matches on TYPE + LOCATION, nearest first) (§1, §7)
$ call POST /match/$REQ                    # "matched" -> Kochi NGO water tanker, ~1 km
$ call GET /resources/$RES                 # availability 2 -> 1, the request is in assigned_requests
$ call POST /match/$REQ                    # again: "skipped" - never takes a second unit

# ---- 2.5 updateStatus(requestId, status) - PUT /requests/{id}/status -> 200            (PDF §1, §5)
$ call PUT /requests/$REQ/status '{"status":"in_progress","note":"tanker on the way"}'
$ call PUT /requests/$REQ/status '{"status":"resolved"}'
$ call GET /resources/$RES                 # unit released: availability back to 2
$ call GET /requests/$REQ | grep -A1 '"status"'      # history: pending -> matched -> in_progress -> resolved

# ---- 2.6 Resource availability updates - PUT /resources/{id}/status -> 200            (PDF §5, §11)
$ call PUT /resources/$RES/status '{"status":"offline"}'          # offline resources are never matched
$ call PUT /resources/$RES/status '{"availability":0}'            # 0 units -> status "busy" automatically
$ call PUT /resources/$RES/status '{"availability":50,"status":"available"}'   # 50 units for the next step

# ---- 2.7 Asynchronous processing through RabbitMQ (several requests at once)          (PDF §7, §10, §11)
# Each request is stored, published to RabbitMQ, consumed by a matching worker, matched, statuses updated.
$ BURST=$(for i in $(seq 1 40); do curl -s -X POST $API/requests -H 'Content-Type: application/json' -d '{"type":"water","details":"burst","location":{"lat":9.9312,"lon":76.2673}}' | python3 -c 'import json,sys; print(json.load(sys.stdin)["id"])' & done; wait); echo $BURST
                                           # 40 requests at the same moment, all accepted at once (201)
$ for i in $(seq 1 14); do queue; sleep 2; done
                                           # watch RabbitMQ: messages WAITING (ready) and being processed
                                           # (unacked) appear, then drain to 0 (its statistics lag ~5 s)
$ for id in $(echo $BURST | cut -d' ' -f1-8); do curl -s $API/requests/$id | python3 scripts/show.py request; done
                                           # "matched" to the tanker, worker=<container id>
$ for id in $BURST; do curl -s $API/requests/$id | python3 scripts/show.py worker; done | sort | uniq -c
                                           # all 40 matched, the work shared by BOTH matching workers
$ curl -s -u erms:$RPASS $RABBIT/api/exchanges/%2F/erms.requests/bindings/source | python3 -m json.tool | grep -E '"(destination|routing_key)"'
                                           # exchange erms.requests -> queue matching.requests (request.#)

# ---- 2.8 Data distribution + shard selection = sha256(key) mod 3                   (PDF §4, §10, §11)
$ for id in $BURST; do curl -s $API/requests/$id | python3 -c 'import json,sys; d=json.load(sys.stdin); print(d["id"], "-> shard", d["_storage"]["shard"], "served by", d["_storage"]["served_by"])'; done
$ ccall GET /internal/shard-of/request:$REQ                       # the coordinator's choice
$ python3 -c "import hashlib,sys; k='request:'+sys.argv[1]; print(k, '-> sha256 mod 3 =', int.from_bytes(hashlib.sha256(k.encode()).digest()[:8],'big') % 3)" $REQ
                                           # same number, computed by hand: the client never chooses a shard
$ cluster                                  # records per shard / per node

# ---- 2.9 Internal service communication (coordinator <-> storage nodes)                    (PDF §6)
$ ccall GET /internal/nodes | grep -E '"node_id"|"status"|"heartbeat_age_s"'   # registered nodes + heartbeats
$ ccall PUT /internal/data/demo:hello '{"value":{"msg":"hi"}}'                 # store  (to every replica)
$ ccall GET /internal/data/demo:hello                                          # retrieve
$ ccall DELETE /internal/data/demo:hello                                       # delete
$ ccall GET /internal/data/demo:hello                                          # 404 now
$ ccall POST /internal/nodes/heartbeat '{"node_id":"nobody","shard_id":0,"url":"http://x"}'   # 404: register first
$ docker compose logs storage-0a | grep -m2 -E "registered with coordinator|SYNC complete"

# ---- 2.10 HTTP status codes: invalid requests, missing items, unavailable services        (PDF §5)
$ call POST /requests '{"type":"pizza","location":{"lat":0,"lon":0}}'       # 422 unknown type
$ call POST /requests '{"type":"water","location":{"lat":123,"lon":0}}'     # 422 latitude out of range
$ call GET /requests/req-000000000000                                       # 404
$ call PUT /requests/$REQ/status '{"status":"pending"}'                      # 409 already resolved
$ call PUT /resources/$RES/status '{}'                                      # 400 nothing to update
$ docker compose stop matching-service
$ call POST /match/$REQ                                                     # 503 matching service unavailable
$ docker compose start matching-service

# ---- 2.11 Storage node failure -> replica -> recovery + resynchronisation          (PDF §8, §10, §11)
# A request that lives on shard 1 (nodes storage-1a + storage-1b; reads normally go to storage-1a):
$ for i in $(seq 1 30); do call POST /requests '{"type":"food","details":"written BEFORE the failure","auto_match":false,"location":{"lat":9.93,"lon":76.26}}' >/dev/null; [ "$(field 'd["_storage"]["shard"]')" = 1 ] && break; done; S1=$(last_id); echo "$S1 is on shard 1"
$ call GET /requests/$S1 | grep served_by                     # BEFORE: served_by storage-1a
$ docker compose stop storage-1a                              # STOP the storage node
$ sleep 8
$ cluster                                                     # DURING: storage-1a "down" (heartbeats stopped)
$ call GET /requests/$S1 | grep served_by                     # DURING: still readable, from the replica storage-1b
$ for i in $(seq 1 30); do call POST /requests '{"type":"food","details":"written DURING the failure","auto_match":false,"location":{"lat":9.93,"lon":76.26}}' >/dev/null; [ "$(field 'd["_storage"]["shard"]')" = 1 ] && break; done; S2=$(last_id); field 'd["_storage"]'
                                                              # DURING: accepted, acked only by storage-1b, degraded true
$ docker compose start storage-1a                             # RESTART it
$ sleep 12
$ cluster                                                     # AFTER: ready again, same record count as storage-1b
$ docker compose logs storage-1a | grep "SYNC complete" | tail -1    # "applied N updates" = the writes it missed
$ call GET /requests/$S2 | grep -E 'served_by|details'        # AFTER: storage-1a serves the write it missed

=====================================================================================================================
 PART 10 - TROUBLESHOOTING
=====================================================================================================================
 "command not found: kubectl / minikube"   -> export PATH="$HOME/.local/bin:$PATH"
 "run ./scripts/init-env.sh first"           -> ./scripts/init-env.sh   (creates / completes .env)
 broker_connected false right after start    -> wait ~20 s (RabbitMQ starting); curl -s localhost:8080/health
 something not running after a reboot        -> docker compose up -d     |   minikube start (pods come back by themselves)
 Kubernetes URLs don't open in Windows       -> python3 scripts/expose.py in a second terminal, use localhost:300xx
 both deployments slow / out of memory        -> run only one: docker compose stop  OR  minikube stop
 "connection refused" on the NodePorts       -> minikube status; kubectl -n erms get pods (wait until all Running)
 a request stays "pending" / "unmatched"     -> no available resource of that type within 50 km: register one;
                                                the sweeper re-queues it automatically
 RabbitMQ / Grafana passwords                -> COMPOSE=1 ./scripts/rabbit-password.sh | ./scripts/grafana-password.sh
                                                (without COMPOSE=1: the Kubernetes Secrets)
